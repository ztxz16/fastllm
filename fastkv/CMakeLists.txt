cmake_minimum_required(VERSION 3.18)
project(FastKVManager VERSION 0.1.0 LANGUAGES CXX)

option(FASTKV_RDMA "Build the native libibverbs transport" OFF)
option(FASTKV_CUDA "Build CUDA memory support" OFF)
option(FASTKV_TESTS "Build standalone tests and diagnostics" ON)
find_package(Threads REQUIRED)
if(UNIX)
  set(FASTKV_NETWORK_SOURCE src/network.cpp)
else()
  set(FASTKV_NETWORK_SOURCE src/network_unavailable.cpp)
endif()
add_library(fastkv STATIC src/core.cpp src/prefix_cache.cpp ${FASTKV_NETWORK_SOURCE} src/rdma.cpp)
set_target_properties(fastkv PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(fastkv PUBLIC cxx_std_17)
target_include_directories(fastkv PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(fastkv PUBLIC Threads::Threads)
if(FASTKV_RDMA)
  find_path(FASTKV_VERBS_INCLUDE infiniband/verbs.h REQUIRED)
  find_library(FASTKV_VERBS_LIBRARY ibverbs REQUIRED)
  target_include_directories(fastkv PRIVATE ${FASTKV_VERBS_INCLUDE})
  target_link_libraries(fastkv PRIVATE ${FASTKV_VERBS_LIBRARY})
  target_compile_definitions(fastkv PRIVATE FASTKV_RDMA=1)
endif()
if(FASTKV_CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_link_libraries(fastkv PRIVATE CUDA::cudart)
  target_compile_definitions(fastkv PRIVATE FASTKV_CUDA=1)
endif()
if(FASTKV_TESTS AND UNIX)
  enable_testing()
  add_executable(fastkv_test test/core_test.cpp)
  target_link_libraries(fastkv_test PRIVATE fastkv)
  add_test(NAME fastkv_core COMMAND fastkv_test)
  add_executable(fastkv_prefix_cache_test test/prefix_cache_test.cpp)
  target_link_libraries(fastkv_prefix_cache_test PRIVATE fastkv)
  add_test(NAME fastkv_prefix_cache COMMAND fastkv_prefix_cache_test)
  add_executable(fastkv_probe tools/probe.cpp)
  target_link_libraries(fastkv_probe PRIVATE fastkv)
endif()
