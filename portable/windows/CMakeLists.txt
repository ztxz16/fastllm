cmake_minimum_required(VERSION 3.23)
project(ftllm_portable_launcher LANGUAGES CXX)
add_executable(ftllm launcher.cpp)
target_compile_features(ftllm PRIVATE cxx_std_17)
target_compile_definitions(ftllm PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET ftllm PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")

add_executable(FastLLM-Launcher WIN32 ../../desktop/windows/entrypoint.cpp)
target_compile_definitions(FastLLM-Launcher PRIVATE FTLLM_GUI_ENTRYPOINT)
add_executable(ftllm-launch-webui ../../desktop/windows/entrypoint.cpp)
add_executable(ftllm-desktop-cli ../../desktop/windows/entrypoint.cpp)
foreach(entrypoint FastLLM-Launcher ftllm-launch-webui ftllm-desktop-cli)
    target_compile_features(${entrypoint} PRIVATE cxx_std_17)
    target_compile_definitions(${entrypoint} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(${entrypoint} PRIVATE shell32 user32)
    set_property(TARGET ${entrypoint} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endforeach()
